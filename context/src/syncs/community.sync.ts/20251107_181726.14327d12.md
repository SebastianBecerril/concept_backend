/**
 * @file src/syncs/community.sync.ts
 * @description This file contains all the synchronizations for the Community concept.
 * It handles API request/response cycles for all community actions using session-based
 * authentication and ensures data integrity by cascading deletions to related concepts.
 */

import {
  Community,
  CommunityBoard,
  Requesting,
  UserAuthentication,
} from "@concepts";
import { actions, Frames, Sync } from "@engine";
import { ID } from "@utils/types.ts";

/** Identifies the user for a given session and returns frames with a `user` binding. */
const bindSessionUser = async (
  frames: Frames,
  session: symbol,
  user: symbol,
) => {
  const enriched: Record<symbol, unknown>[] = [];
  for (const frame of frames) {
    const sessionId = frame[session] as ID | undefined;
    if (!sessionId) continue;
    const matches = await UserAuthentication._getUserForSession({ sessionId });
    for (const match of matches) {
      enriched.push({ ...frame, [user]: match.user });
    }
  }
  return enriched.length > 0 ? new Frames(...enriched) : new Frames();
};

export const CommunityCreateRequest: Sync = (
  { request, name, description, session, creator },
) => ({
  when: actions(
    [Requesting.request, {
      path: "/Community/createCommunity",
      name,
      description,
      session,
    }, { request }],
  ),
  where: (frames) => bindSessionUser(frames, session, creator),
  then: actions(
    [Community.createCommunity, { name, description, creator }],
  ),
});

export const CommunityGetCommunityById: Sync = (
  { request, community, communityRecord },
) => ({
  when: actions(
    [Requesting.request, { path: "/Community/_getCommunityById", community }, {
      request,
    }],
  ),
  where: async (frames) => {
    const enriched: Record<symbol, unknown>[] = [];
    for (const frame of frames) {
      const communityId = frame[community] as ID | undefined;
      if (!communityId) continue;
      const record = await Community._getCommunityById({
        community: communityId,
      });
      enriched.push({ ...frame, [communityRecord]: record });
    }
    return enriched.length > 0 ? new Frames(...enriched) : new Frames();
  },
  then: actions([Requesting.respond, { request, community: communityRecord }]),
});

export const CascadeCommunityDeletionToPosts: Sync = (
  { community, requester, post },
) => ({
  when: actions(
    [Community.deleteCommunity, { community, requester }, {}],
  ),
  where: async (frames) => {
    const enriched: Record<symbol, unknown>[] = [];
    for (const frame of frames) {
      const communityId = frame[community] as ID | undefined;
      if (!communityId) continue;
      const posts = await CommunityBoard._getPostsByCommunity({
        community: communityId,
      });
      for (const record of posts) {
        enriched.push({ ...frame, [post]: record._id });
      }
    }
    return enriched.length > 0 ? new Frames(...enriched) : new Frames();
  },
  then: actions(
    [CommunityBoard.deletePost, { post, requester }],
  ),
});
