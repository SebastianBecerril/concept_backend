export const CommunityCreateRequest: Sync = (
  { request, name, description, creator },
) => ({
  when: actions(
    [Requesting.request, {
      path: "/Community/createCommunity",
      name,
      description,
      creator,
    }, { request }],
  ),
  then: actions(
    [Community.createCommunity, { name, description, creator }],
  ),
});

const frameWith = (
  frames: Frames,
  updates: Record<symbol, unknown>,
): Frames => {
  const seed = frames[0] ?? {};
  return new Frames({ ...seed, ...updates });
};

export const CommunityGetCommunityById: Sync = (
  { request, community, communityRecord },
) => ({
  when: actions(
    [Requesting.request, { path: "/Community/_getCommunityById", community }, {
      request,
    }],
  ),
  where: async (frames) => {
    const record = await Community._getCommunityById({ community });
    return frameWith(frames, { [communityRecord]: record });
  },
  then: actions([Requesting.respond, { request, community: communityRecord }]),
});

export const CascadeCommunityDeletionToPosts: Sync = (
  { community, requester, post },
) => ({
  when: actions(
    [Community.deleteCommunity, { community, requester }, {}],
  ),
  where: async (frames) => {
    const posts = await CommunityBoard._getPostsByCommunity({ community });
    if (posts.length === 0) {
      return new Frames();
    }
    return new Frames(
      ...posts.map((record) => ({
        [post]: record._id,
        [requester]: requester,
      })),
    );
  },
  then: actions(
    [CommunityBoard.deletePost, { post, requester }],
  ),
});
